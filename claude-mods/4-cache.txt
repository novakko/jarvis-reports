## What you are building

**Name on disk:** `cache-clock` (plugin name, command `/cache-clock`).

**What it does.** A band above the prompt shows whether the prompt cache is still warm and what letting it go cold would cost.

**Band:** chip ` ● CACHE ` (bold, `#0b0b0f` on `#22d3ee`), bold status `warm 42m` or `COLD`, then `▰▰▰▰▰▱ 94% reused · 180k cached · cold rewrite ≈ $1.80 (est.)` and a `Compact` button. Status colours: green over 15 minutes left, amber at 15 minutes or fewer, red at cold. The band is hidden until the first cache activity, during a survey, or when off. It stacks with other mods' bands.

**Rules:**
- The clock restarts only on a main-chat model step that read or wrote the cache. Subagent steps count in the totals but never restart the clock.
- Cache lifetime is the `cacheMinutes` option in the `/config` menu: `5` or `60`, default `60`.
- Reuse percent is the last main step's reads divided by reads + writes + input, rounded. The bar has six cells, one per sixth, rounded down.
- Cold rewrite is cached tokens × input price × write multiplier (1.25 for 5 minutes, 2 for 60 minutes). Saved is read tokens × input price × (1 − read multiplier), and appears only in `/cache-clock status`. Haiku 5.5 uses $0.50 instead of $0.10 past 100k tokens.
- Prices are in `hooks/lib.ts` and cover Claude Opus, Sonnet and Haiku only. A model not listed shows `≈ ?`, never a guess. Its read tokens are counted separately.
- Timer every 30 seconds. One toast per cache write when 5 minutes are left, only with the 60-minute lifetime: `Cache goes cold in 5 min: 180k tokens, rewrite ≈ $1.80 (estimate). Send anything to keep it warm.`
- The Compact button compacts the chat and toasts `Compacted: 180k → 9k tokens`, `Compact skipped: <why>` or `Compact failed: <why>`.

**Command:** `/cache-clock on | off | status`. No argument means `status`. On and off last for the session only, and are not remembered. `/clear` does not reset the band; it updates at the next cache step.

import type { CacheState } from '../types'

export const EMPTY: CacheState = {
  lastCacheAt: null,
  model: null,
  cachedTokens: 0,
  lastReads: 0,
  lastWrites: 0,
  lastInput: 0,
  totalReads: 0,
  totalWrites: 0,
  totalInput: 0,
  savedUsd: 0,
  unpricedReads: 0,
}

export const MINUTE = 60_000
export const AMBER_AT_MS = 15 * MINUTE
export const WARN_AT_MS = 5 * MINUTE

type Price = {
  /** Base input, $ per million tokens. */
  input: number
  /** Cache-read price as a fraction of the base input price. */
  readMult: number
  /** Base input past 100K prompt tokens, where the model prices it differently. */
  longInput?: number
}

// $ per million input tokens, from the claude-api skill's "Current Models"
// table (cached 2026-10-06). Cache reads are 0.1x base unless noted in its
// shared/prompt-caching.md: 0.05x on Opus 5.5. Cache writes are 1.25x base for the 5-minute cache, 2x for the
// 1-hour cache. A model missing here has no price: never estimate one.
const PRICES: Readonly<Record<string, Price>> = {
  'claude-opus-5-5': { input: 4, readMult: 0.05 },
  'claude-opus-5': { input: 5, readMult: 0.1 },
  'claude-opus-4-8': { input: 5, readMult: 0.1 },
  'claude-opus-4-7': { input: 5, readMult: 0.1 },
  'claude-opus-4-6': { input: 5, readMult: 0.1 },
  'claude-sonnet-5-5': { input: 2, readMult: 0.1 },
  'claude-sonnet-5': { input: 2, readMult: 0.1 },
  'claude-sonnet-4-6': { input: 3, readMult: 0.1 },
  'claude-haiku-5-5': { input: 0.1, readMult: 0.1, longInput: 0.5 },
  'claude-haiku-4-5': { input: 1, readMult: 0.1 },
}

/** The price row for a model id: exact, or the id plus a `-YYYYMMDD` snapshot date. */
export function priceFor(model: string | null): Price | null {
  if (model === null) {
    return null
  }

  for (const [id, price] of Object.entries(PRICES)) {
    if (model === id || (model.startsWith(`${id}-`) && /^\d{8}$/.test(model.slice(id.length + 1)))) {
      return price
    }
  }

  return null
}

function inputPrice(price: Price, promptTokens: number): number {
  return price.longInput !== undefined && promptTokens > 100_000 ? price.longInput : price.input
}

export function writeMultiplier(ttlMinutes: number): number {
  return ttlMinutes >= 60 ? 2 : 1.25
}

/** What re-writing `cachedTokens` into a cold cache costs, or null for an unpriced model. */
export function coldRewriteUsd(model: string | null, cachedTokens: number, ttlMinutes: number): number | null {
  const price = priceFor(model)

  return price === null ? null : (cachedTokens * inputPrice(price, cachedTokens) * writeMultiplier(ttlMinutes)) / 1e6
}

/** What reading `reads` tokens from cache saved against full-price input, or null if unpriced. */
export function savedUsd(model: string | null, reads: number): number | null {
  const price = priceFor(model)

  return price === null ? null : (reads * inputPrice(price, reads) * (1 - price.readMult)) / 1e6
}

export function formatUsd(usd: number | null): string {
  return usd === null ? '≈ ?' : `≈ $${usd.toFixed(2)}`
}

export function formatTokens(n: number): string {
  if (n >= 1_000_000) {
    return `${(n / 1_000_000).toFixed(1)}M`
  }

  return n >= 1000 ? `${Math.round(n / 1000)}k` : String(n)
}

/** Cache reads over everything the request processed, as a whole percent. */
export function pctReused(reads: number, writes: number, input: number): number {
  const total = reads + writes + input

  return total === 0 ? 0 : Math.round((reads / total) * 100)
}

/** Six cells, one per sixth reused: 94% fills five. */
export function bar(pct: number): string {
  const filled = Math.min(6, Math.floor((pct / 100) * 6))

  return '▰'.repeat(filled) + '▱'.repeat(6 - filled)
}

/** Milliseconds of warmth left, 0 once cold, null before any cache activity. */
export function remainingMs(lastCacheAt: number | null, ttlMs: number, now: number): number | null {
  return lastCacheAt === null ? null : Math.max(0, lastCacheAt + ttlMs - now)
}

export type Tone = 'green' | 'amber' | 'red'

export function toneFor(remaining: number): Tone {
  if (remaining === 0) {
    return 'red'
  }

  return remaining <= AMBER_AT_MS ? 'amber' : 'green'
}

export const COLORS: Readonly<Record<Tone, string>> = {
  green: '#4ade80',
  amber: '#fbbf24',
  red: '#f87171',
}

/**
 * True once, per cache write, when the clock first reaches the last 5 minutes.
 * A 5-minute cache is always inside them, so it never warns.
 */
export function shouldWarn(
  remaining: number | null,
  ttlMs: number,
  lastCacheAt: number | null,
  warnedFor: number | null,
): boolean {
  return (
    remaining !== null &&
    ttlMs > WARN_AT_MS &&
    remaining > 0 &&
    remaining <= WARN_AT_MS &&
    warnedFor !== lastCacheAt
  )
}

import { atom, read, update } from 'claude-code'
import type { Register } from 'claude-code'

import {
  COLORS,
  EMPTY,
  MINUTE,
  bar,
  coldRewriteUsd,
  formatTokens,
  formatUsd,
  pctReused,
  remainingMs,
  savedUsd,
  shouldWarn,
  toneFor,
} from './lib'

const cache = atom({ plugin: 'cache-clock', key: 'cache' } as const, EMPTY)
const isOn = atom({ plugin: 'cache-clock', key: 'isOn' } as const, true)
const now = atom({ plugin: 'cache-clock', key: 'now' } as const, 0)
const warnedFor = atom({ plugin: 'cache-clock', key: 'warnedFor' } as const, null)

const TICK_MS = 30_000

export const register: Register = (on, options) => {
  const ttlMinutes = String(options.cacheMinutes) === '5' ? 5 : 60
  const ttlMs = ttlMinutes * MINUTE

  on('session.start', async ($, e, next) => {
    await $.command.register({
      name: 'cache-clock',
      description: 'Prompt cache clock',
      argumentHint: 'on | off | status',
    })

    $.clock.every(TICK_MS, async () => {
      const at = await $.clock.now()
      await update($, now, () => at)

      const state = await read($, cache)
      const remaining = remainingMs(state.lastCacheAt, ttlMs, at)

      if (!(await read($, isOn)) || !shouldWarn(remaining, ttlMs, state.lastCacheAt, await read($, warnedFor))) {
        return
      }

      await update($, warnedFor, () => state.lastCacheAt)
      $.ui.toast(
        `Cache goes cold in 5 min: ${formatTokens(state.cachedTokens)} tokens, ` +
          `rewrite ${formatUsd(coldRewriteUsd(state.model, state.cachedTokens, ttlMinutes))} (estimate). ` +
          'Send anything to keep it warm.',
      )
    })

    return next(e)
  })

  on('turn.step', async function* ($, e, next) {
    const response = yield* next(e)
    const usage = response.usage

    if (usage === null) {
      return response
    }

    const reads = usage.cache_read_input_tokens
    const writes = usage.cache_creation_input_tokens
    const input = usage.input_tokens
    const isMain = e.agentId === undefined
    const isCacheStep = isMain && reads + writes > 0
    const at = await $.clock.now()
    const saved = savedUsd(usage.model, reads)

    await update($, cache, c => ({
      ...c,
      totalReads: c.totalReads + reads,
      totalWrites: c.totalWrites + writes,
      totalInput: c.totalInput + input,
      savedUsd: c.savedUsd + (saved ?? 0),
      unpricedReads: c.unpricedReads + (saved === null ? reads : 0),
      ...(isMain ? { lastReads: reads, lastWrites: writes, lastInput: input } : {}),
      ...(isCacheStep ? { lastCacheAt: at, model: usage.model, cachedTokens: reads + writes } : {}),
    }))

    if (isCacheStep) {
      await update($, now, () => at)
    }

    return response
  })

  on('command.run', { command: 'cache-clock' }, async ($, e) => {
    const arg = e.args.trim().toLowerCase()

    if (arg === 'on' || arg === 'off') {
      await update($, isOn, () => arg === 'on')

      return { text: `cache-clock ${arg}` }
    }

    if (arg !== '' && arg !== 'status') {
      return { text: 'Usage: /cache-clock on | off | status' }
    }

    const state = await read($, cache)
    const remaining = remainingMs(state.lastCacheAt, ttlMs, await $.clock.now())
    const left =
      remaining === null ? 'no cache activity yet' : remaining === 0 ? 'COLD' : `warm, ${Math.ceil(remaining / MINUTE)}m left`
    const missed =
      state.unpricedReads > 0 ? ` (${formatTokens(state.unpricedReads)} read tokens on models without a price not counted)` : ''

    return {
      text: [
        `cache-clock ${(await read($, isOn)) ? 'on' : 'off'} · cache ${left} of ${ttlMinutes}m`,
        `reads ${formatTokens(state.totalReads)} · writes ${formatTokens(state.totalWrites)} · fresh input ${formatTokens(state.totalInput)} (all steps, subagents included)`,
        `${pctReused(state.lastReads, state.lastWrites, state.lastInput)}% reused on the last main step`,
        `saved ${formatUsd(state.savedUsd)} (estimate)${missed}`,
      ].join('\n'),
    }
  })

  on('ui.render', { component: 'AbovePrompt' }, async ($, e, next) => {
    const state = await read($, cache)
    await read($, now)

    if (e.props.hasSurvey || state.lastCacheAt === null || !(await read($, isOn))) {
      return next(e)
    }

    const below = await next(e)
    const remaining = remainingMs(state.lastCacheAt, ttlMs, await $.clock.now()) ?? 0
    const tone = toneFor(remaining)
    const pct = pctReused(state.lastReads, state.lastWrites, state.lastInput)
    const rewrite = formatUsd(coldRewriteUsd(state.model, state.cachedTokens, ttlMinutes))
    const status = remaining === 0 ? 'COLD' : `warm ${Math.ceil(remaining / MINUTE)}m`
    const { Box, Button, Text } = $.ui.resolve(e)

    const compact = async () => {
      try {
        const result = await $.session.compact()

        $.ui.toast(
          'skip' in result
            ? `Compact skipped: ${result.skip}`
            : `Compacted: ${formatTokens(result.tokensBefore ?? 0)} → ${formatTokens(result.tokensAfter ?? 0)} tokens`,
        )
      } catch (error) {
        $.ui.toast(`Compact failed: ${error instanceof Error ? error.message : String(error)}`)
      }
    }

    const line = (
      <Box>
        <Text bold color="#0b0b0f" backgroundColor="#22d3ee">
          {' ● CACHE '}
        </Text>
        <Text bold color={COLORS[tone]}>
          {` ${status} `}
        </Text>
        <Text>
          {`${bar(pct)} ${pct}% reused · ${formatTokens(state.cachedTokens)} cached · cold rewrite ${rewrite} (est.) `}
        </Text>
        <Button key="compact" label="Compact" onPress={compact} />
      </Box>
    )

    return below ? (
      <Box flexDirection="column">
        {line}
        {below}
      </Box>
    ) : (
      line
    )
  })
}

const START = 1_000_000
const MINUTE = 60_000

const BAND = {
  plugin: 'cache-clock',
  component: 'AbovePrompt',
  requestId: 'band',
  surface: 'terminal',
  props: {
    hasSurvey: false,
    isWorking: false,
    maxRows: 10,
    bodyColumns: 120,
    scroll: { offset: 0, bodyRows: 10 },
    view: {},
  },
} as const

const CACHE = { plugin: 'cache-clock', key: 'cache' } as const

type Step = { agentId?: string; reads: number; writes: number; input: number; model?: string }

// The engine beneath the plugin: it answers every model request with `step`.
function stepWith(on: On, step: () => Step): void {
  on('turn.step', async function* (_, e) {
    const s = step()

    return {
      turnId: e.turnId,
      index: e.index,
      answer: '',
      toolUses: [],
      stopReason: 'end_turn',
      usage: {
        model: s.model ?? 'claude-opus-4-8',
        input_tokens: s.input,
        output_tokens: 10,
        cache_read_input_tokens: s.reads,
        cache_creation_input_tokens: s.writes,
      },
    }
  })
}

async function runStep($: Engine, agentId?: string): Promise<void> {
  const stream = $.turn.step({ turnId: 't', index: 0, model: 'm', messageCount: 1, agentId })

  for await (const _chunk of stream) {
    // drained: the hooks run as the chunks pass
  }
  await stream.result
}

const slash = (args: string) =>
  ({
    command: 'cache-clock',
    args,
    origin: { kind: 'composer' },
    presentation: { isFullscreen: false, columns: 80 },
  }) as const

function engineBeneath(on: On, toasts: string[] = []) {
  const clock = mock.clock(on, { now: START })

  on('command.register', () => ({ value: { command: 'cache-clock' } }))
  on('session.start', (_, e) => ({ cwd: e.cwd }))
  on('ui.toast', (_, e) => {
    toasts.push(e.text)

    return { value: undefined }
  })
  // Another mod's band line, already in the chain beneath ours.
  on('ui.render', { component: 'AbovePrompt' }, ($, e) => {
    const { Text } = $.ui.resolve(e)

    return <Text>other mod line</Text>
  })

  return clock
}

test('only main-loop cache steps restart the countdown; subagents count in totals', async ($, on) => {
  const clock = engineBeneath(on)
  let step: Step = { reads: 0, writes: 0, input: 0 }
  stepWith(on, () => step)

  const status = async (): Promise<string> =>
    (await $.command.run(slash('status'))).text ?? ''

  await runStep($)
  expect(await status()).toContain('no cache activity yet')

  step = { reads: 90_000, writes: 10_000, input: 0 }
  await runStep($)
  expect(await status()).toContain('warm, 60m left')

  await clock.advance(10 * MINUTE)
  step = { agentId: 'sub1', reads: 50_000, writes: 0, input: 0 }
  await runStep($, 'sub1')
  const afterSub = await status()
  expect(afterSub).toContain('warm, 50m left')
  expect(afterSub).toContain('reads 140k')
  expect(afterSub).toContain('90% reused on the last main step')

  step = { reads: 100_000, writes: 0, input: 0 }
  await runStep($)
  expect(await status()).toContain('warm, 60m left')
})
